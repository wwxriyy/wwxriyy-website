-- Single background color for the whole UI (keep in sync with wezterm.lua)
local BG = "#141414"

local function unify_ui()
    local function set_bg(name)
        local ok, hl = pcall(vim.api.nvim_get_hl, 0, { name = name, link = false })
        if not ok or not hl then
            return
        end
        hl.bg = BG
        vim.api.nvim_set_hl(0, name, hl)
    end

    local groups = {
        "Normal", "NormalNC", "SignColumn", "FoldColumn", "LineNr", "CursorLineNr",
        "EndOfBuffer", "StatusLine", "StatusLineNC", "WinBar", "WinBarNC",
        "WinSeparator", "VertSplit",
        "NeoTreeNormal", "NeoTreeNormalNC", "NeoTreeEndOfBuffer",
        "NeoTreeStatusLine", "NeoTreeWinSeparator",
        "TreesitterContext", "TreesitterContextLineNumber",
        "GitSignsAdd", "GitSignsChange", "GitSignsDelete",
        "GitSignsTopdelete", "GitSignsChangedelete", "GitSignsUntracked",
        "DiagnosticSignError", "DiagnosticSignWarn", "DiagnosticSignInfo",
        "DiagnosticSignHint", "DiagnosticSignOk",
    }
    for _, name in ipairs(groups) do
        set_bg(name)
    end
    for _, name in ipairs(vim.fn.getcompletion("BufferLine", "highlight")) do
        set_bg(name)
    end

    vim.api.nvim_set_hl(0, "WinSeparator", { fg = "#2a2a2a", bg = BG })
    vim.api.nvim_set_hl(0, "NeoTreeWinSeparator", { fg = "#2a2a2a", bg = BG })
end

vim.api.nvim_create_autocmd({ "ColorScheme", "VimEnter" }, {
    callback = function()
        vim.schedule(unify_ui)
    end,
})
vim.api.nvim_create_autocmd("BufWinEnter", {
    once = true,
    callback = function()
        vim.defer_fn(unify_ui, 100)
    end,
})
